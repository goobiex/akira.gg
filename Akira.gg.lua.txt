local function finite(value)
    return type(value) == "number" and value == value and math.abs(value) < math.huge
end

local function cleanName(value)
    if type(value) ~= "string" then return nil end
    local name = value:match("^%s*(.-)%s*$")
    if #name < 1 or #name > 32 or not name:match("^[%w _%.%-]+$") then return nil end
    return name
end

local function copy(value)
    if type(value) ~= "table" then return value end
    local result = {}
    for key, child in pairs(value) do result[key] = copy(child) end
    return result
end

local function valueFor(entry, value)
    if type(value) ~= type(entry.default) then return entry.default end
    if type(entry.min) == "table" then return table.find(entry.min, value) and value or entry.default end
    if type(value) == "number" then
        return finite(value) and math.clamp(value, entry.min, entry.max) or entry.default
    end
    if entry.key == "Color" then return value:match("^#%x%x%x%x%x%x$") and value:upper() or entry.default end
    if entry.key == "CombatKey" or entry.key == "MenuKey" then
        for _, key in ipairs(Enum.KeyCode:GetEnumItems()) do
            if key.Name == value and key ~= Enum.KeyCode.Unknown then return value end
        end
        return entry.default
    end
    return value
end

local function normalize(source, weaponsOnly)
    source = type(source) == "table" and source or {}
    local result = {}
    for _, entry in ipairs(fields) do
        if (entry.page == "Weapons") == weaponsOnly then
            result[entry.key] = valueFor(entry, source[entry.key])
        end
    end
    if not weaponsOnly then
        result.Weapons = {}
        local count = 0
        if type(source.Weapons) == "table" then
            for key, value in pairs(source.Weapons) do
                if cleanName(key) == key and type(value) == "table" and count < 32 then
                    result.Weapons[key] = normalize(value, true)
                    count = count + 1
                end
            end
        end
    end
    return result
end

local function decode(payload)
    if type(payload) ~= "string" or #payload > 95000 then return nil end
    local ok, data = pcall(JSON.JSONDecode, JSON, payload)
    return ok and type(data) == "table" and data or nil
end

local function restore(payload)
    local data = decode(payload)
    if not data or data.version ~= 1 or type(data.profiles) ~= "table" then return end
    local count = 3
    for name, settings in pairs(data.profiles) do
        if cleanName(name) == name and type(settings) == "table" and (profiles[name] or count < 12) then
            if not profiles[name] then count = count + 1 end
            profiles[name] = normalize(settings, false)
        end
    end
    if type(data.active) == "string" and profiles[data.active] then activeName = data.active end
    config = copy(profiles[activeName])
end

local function changed(selectionOnly)
    if not selectionOnly then profileDirty = true end

local function save(explicit)
    if explicit then
        revision = revision + 1
        due = os.clock()
    end
    if saving or revision == savedRevision then return end
    if explicit or profileDirty then
        profiles[activeName] = copy(config)
        profileDirty = false
    end
    local snapshot = revision
    local payload = JSON:JSONEncode({version = 1, active = activeName, profiles = profiles})
    if #payload > 95000 then
        due = math.huge
        message("Profile collection exceeds the storage limit")
        return
    end
    playerGui:SetAttribute("AkiraProfilesV1", payload)
    if not profileRemote then
        savedRevision = snapshot
        if explicit then message("Saved for this session") end
        return
    end
    saving = true
    task.spawn(function()
        local ok, response = pcall(profileRemote.InvokeServer, profileRemote, "save", payload)
        saving = false
        if not alive then return end
        if ok and response == true then
            savedRevision = snapshot
            if explicit then message("Profiles saved") end
        else
            due = os.clock() + 15
            message("Server save failed · session copy retained")
        end
    end)
end

local function eligible(model, owner, character)
    if model == character or not model:IsDescendantOf(workspace) then return false end
    local match = character:GetAttribute("AkiraMatch")
    if match ~= model:GetAttribute("AkiraMatch") then return false end
    local team = character:GetAttribute("AkiraTeam")
    if team ~= nil then return team ~= model:GetAttribute("AkiraTeam") end
    return not owner or owner.Neutral or player.Neutral or owner.Team ~= player.Team
end

local function sample(model, owner, character, now)
    if not eligible(model, owner, character) then return end
    local root = rootOf(model)
    local humanoid = model:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid or humanoid.Health <= 0 then return end
    local item = observations[model]
    local velocity = root.AssemblyLinearVelocity
    if not item then
        item = {velocity = velocity, acceleration = Vector3.zero, evasion = 0, exposures = 0,
            visible = false, lastExposure = 0, exposureMean = 0, damage = 0, lastDamage = 0}
        observations[model] = item
    end
    if item.time then
        local dt = now - item.time
        if dt > 0 then
            local accel = (velocity - item.velocity) / dt
            if accel.Magnitude > 240 then accel = accel.Unit * 240 end
            item.acceleration = item.acceleration:Lerp(accel, 1 - math.exp(-dt * 8))
            local flat = Vector3.new(accel.X, 0, accel.Z)
            item.evasion = item.evasion + (flat.Magnitude - item.evasion) * (1 - math.exp(-dt * 3))
        end
    end
    item.model, item.owner, item.root, item.humanoid = model, owner, root, humanoid
    item.velocity, item.time = velocity, now
    table.insert(opponents, item)
end

local function updateOpponents(now)
    table.clear(opponents)
    local char = player.Character
    if char then
        for _, owner in ipairs(Players:GetPlayers()) do
            if owner ~= player and owner.Character then sample(owner.Character, owner, char, now) end
        end
        for model in pairs(tagged) do
            if not Players:GetPlayerFromCharacter(model) then sample(model, nil, char, now) end
        end
    end
    for model, item in pairs(observations) do
        if item.time ~= now then observations[model] = nil end
    end
end

local function trackVisibility(item, visible, now)
    if visible == item.visible then return end
    item.visible = visible
    if visible then
        item.lastExposure = now
    elseif item.lastExposure > 0 then
        local duration = now - item.lastExposure
        if duration >= 0.06 then
            item.exposures = item.exposures + 1
            item.exposureMean = item.exposures == 1 and duration or item.exposureMean * 0.7 + duration * 0.3
        end
    end
end

local function counterFor(item)
    if config.Mode ~= "Rage" then return 1, 0, "Default" end
    local mode = config.Counter
    if mode == "Adaptive" then
        if item.evasion >= config.CustomThreshold then
            mode = "UE"
        elseif item.exposures >= 2 and item.exposureMean < 0.55 then
            mode = "Sleepy"
        else
            mode = "Default"
        end
    end
    if mode == "UE" then return 0.55, 0.04, mode end
    if mode == "Sleepy" then return 0.85, 0.13, mode end
    if mode == "Custom" then return config.CustomLead, config.CustomDelay, mode end
    return 1, 0.025, mode
end

local function intercept(relative, velocity, speed)
    local a = velocity:Dot(velocity) - speed * speed
    local b = 2 * relative:Dot(velocity)
    local c = relative:Dot(relative)
    if c < 0.0001 then return 0 end
    if math.abs(a) < 0.0001 then
        if b < 0 then return -c / b end
        return nil
    end
    local discriminant = b * b - 4 * a * c
    if discriminant < 0 then return nil end
    local square = math.sqrt(discriminant)
    local q = -0.5 * (b + (b >= 0 and square or -square))
    local first, second = q / a, q ~= 0 and c / q or -1
    if first > 0 and second > 0 then return math.min(first, second) end
    if first > 0 then return first end
    if second > 0 then return second end
    return nil
end

local function predicted(item, position, origin, weapon, scale)
    local rage = config.Mode == "Rage"
    local lead = (rage and config.Lead or 1) * weapon.WeaponLead * scale
    local velocity = item.velocity * lead
    local latency = rage and config.Latency or 0
    if weapon.Speed == 0 then return position + velocity * latency end
    local time = intercept(position - origin + velocity * latency, velocity, weapon.Speed)
    if not time or time > 3 then return nil end
    local accel = rage and item.acceleration * config.Acceleration * lead or Vector3.zero
    local gravity = Vector3.new(0, workspace.Gravity * weapon.Gravity, 0)
    local aim
    for pass = 1, 4 do
        local travel = time + latency
        aim = position + velocity * travel + accel * (0.5 * travel * travel) + gravity * (0.5 * time * time)
        local result = (aim - origin).Magnitude / weapon.Speed
        if result > 3 then return nil end
        if math.abs(result - time) < 0.005 or pass == 4 then
            if math.abs(result - time) <= 0.06 then return aim end
            return nil
        end
        time = result
    end
    return aim
end

local function bodyPoints(item, setting, camera, center)
    local head = item.model:FindFirstChild("Head")
    local torso = item.model:FindFirstChild("UpperTorso") or item.model:FindFirstChild("Torso")
    head = head and head:IsA("BasePart") and head or item.root
    torso = torso and torso:IsA("BasePart") and torso or item.root
    if setting == "Head" then return {head} end
    if setting == "Torso" then return {torso} end
    if setting == "Root" then return {item.root} end
    local candidates = {head}
    if torso ~= head then table.insert(candidates, torso) end
    if item.root ~= head and item.root ~= torso then table.insert(candidates, item.root) end
    table.sort(candidates, function(a, b)
        local av = camera:WorldToViewportPoint(a.Position)
        local bv = camera:WorldToViewportPoint(b.Position)
        return (Vector2.new(av.X, av.Y) - center).Magnitude < (Vector2.new(bv.X, bv.Y) - center).Magnitude
    end)
    return candidates
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude
rayParams.IgnoreWater = true
local function clearPath(origin, position, model)
    local direction = position - origin
    if direction.Magnitude < 0.01 then return true end
    local hit = workspace:Raycast(origin, direction, rayParams)
    return not hit or hit.Instance:IsDescendantOf(model)
end

local function radiusFor(weapon)
    if weapon.WeaponFov > 0 then return weapon.WeaponFov end
    return config.Mode == "Rage" and config.RageFov or config.LegitFov
end

local function pathClear(origin, point, item, weapon)
    if weapon.Speed == 0 or weapon.Gravity == 0 then
        return clearPath(origin, point, item.model)
    end
    local offset = point - origin
    local flight = offset.Magnitude / weapon.Speed
    local drop = Vector3.new(0, workspace.Gravity * weapon.Gravity * flight * flight * 0.5, 0)
    local previousPoint = origin
    for step = 1, 4 do
        local alpha = step / 4
        local nextPoint = origin + offset * alpha - drop * alpha * alpha
        if not clearPath(previousPoint, nextPoint, item.model) then return false end
        previousPoint = nextPoint
    end
    return true
end

local function acquire(camera, origin, weapon, now, tracking)
    local center = camera.ViewportSize / 2
    local radius = radiusFor(weapon)
    local range = math.min(config.Range, weapon.WeaponRange)
    local best, bestScore = nil, math.huge
    local body = weapon.WeaponBody == "Profile" and config.Body or weapon.WeaponBody
    local fullView = config.Mode == "Rage" and config.FullView
    rayParams.FilterDescendantsInstances = player.Character and {player.Character, camera} or {camera}
    for _, item in ipairs(opponents) do
        local distance = (item.root.Position - origin).Magnitude
        local visible, evaluated = false, false
        if item.root.Parent and item.humanoid.Health > 0 and distance <= range
            and item.model:GetAttribute("AkiraProtected") ~= true then
            local lead, settle, counter = counterFor(item)
            for _, part in ipairs(bodyPoints(item, body, camera, center)) do
                local point = predicted(item, part.Position, origin, weapon, lead)
                if point then
                    local projected, inView = camera:WorldToViewportPoint(point)
                    local offset = (Vector2.new(projected.X, projected.Y) - center).Magnitude
                    if fullView or (inView and offset <= radius) then
                        evaluated = true
                        visible = pathClear(origin, point, item, weapon)
                        if visible or not config.Visible then
                            local screenScore = inView and offset / radius or 2 + distance / range
                            local score = screenScore
                            if config.Priority == "Distance" then score = distance / range + screenScore * 0.2 end
                            if config.Priority == "Health" then
                                score = item.humanoid.Health / math.max(1, item.humanoid.MaxHealth) + screenScore * 0.2
                            end
                            if config.Priority == "Threat" then
                                local recent = item.damage * math.exp(-(now - item.lastDamage) / 6)
                                score = distance / range + screenScore * 0.25 - math.min(1.5, recent / 100)
                            end
                            if target and item == target.item then score = score - config.Lock * 0.15 end
                            if not visible then score = score + 0.4 end
                            if score < bestScore then
                                bestScore = score
                                best = {item = item, part = part, point = point, visible = visible,
                                    settle = settle, counter = counter, distance = distance}
                            end
                            break
                        end
                    end
                end
            end
        end
        if tracking then trackVisibility(item, evaluated and visible, now) end
    end
    return best
end

local held, toggled, touchAim = false, false, false

local random = Random.new()
resolve.OnInvoke = function(origin, direction)
    if typeof(origin) ~= "Vector3" or typeof(direction) ~= "Vector3" then return direction end
    local length = direction.Magnitude
    local camera = workspace.CurrentCamera
    if not finite(origin.Magnitude) or not finite(length) or length < 0.0001 or not camera or not armed() then
        return direction
    end
    if config.Mode == "Legit" and (not config.Silent or random:NextNumber(0, 100) >= config.Chance) then
        return direction
    end
    local weapon = weaponSettings()
    if not weapon.Enabled then return direction end
    local solution = acquire(camera, origin, weapon, os.clock(), false)
    local offset = solution and solution.point - origin
    return offset and offset.Magnitude > 0.001 and offset.Unit * length or direction
end

connect(report.Event, function(kind, model, amount)
    local item = typeof(model) == "Instance" and observations[model] or nil
    local damage = finite(amount) and math.clamp(amount, 0, 10000) or 0
    if kind == "Shot" then
        accuracy.shots = accuracy.shots + 1
    elseif kind == "Hit" then
        accuracy.hits = accuracy.hits + 1
        accuracy.damage = accuracy.damage + damage
    elseif kind == "DamageTaken" then
        accuracy.received = accuracy.received + damage
        if item then
            local now = os.clock()
            item.damage = item.damage * math.exp(-(now - item.lastDamage) / 6) + damage
            item.lastDamage = now
        end
    end
end)

local full = {Size = UDim2.fromScale(1, 1)}
local function setting(entry)
    local card, area = row(entry)
    local weapon = entry.page == "Weapons"
    local key = entry.key
    local function get()
        local set = weapon and (config.Weapons[editingWeapon] or config.Weapons.Default or defaultWeapon) or config
        return set[key]
    end
    local function set(value)
        local settings = config
        if weapon then
            if not config.Weapons[editingWeapon] then
                local count = 0
                for _ in pairs(config.Weapons) do count = count + 1 end
                if count >= 32 then message("This profile has 32 weapon configurations") return end
                config.Weapons[editingWeapon] = copy(config.Weapons.Default or defaultWeapon)
                changed()
            end
            settings = config.Weapons[editingWeapon]
        end
        local normalized = valueFor(entry, value)
        if settings[key] ~= normalized then
            settings[key] = normalized
            changed()
        end
    end
    if type(entry.min) == "table" or key == "Color" then
        local choices = entry.min or {"#72EAD1", "#F59BA7", "#F4D580", "#94B8FF", "#FFFFFF", "#C7A4FA"}
        local expanded = false
        local pop
        local dropdown = button(area, get() .. "  ▾", full, function()
            expanded = not expanded
            if pop then pop:Destroy() pop = nil end
            local initial = mobile and 95 or 58
            card.Size = UDim2.new(1, -8, 0, expanded and initial + #choices * 34 or initial)
            if not expanded then return end
            pop = new("Frame", card, {
                Size = UDim2.new(1, -20, 0, #choices * 34), Position = UDim2.fromOffset(10, initial),
                BackgroundTransparency = 1,
            })
            for index, option in ipairs(choices) do
                local tint = key == "Color" and Color3.fromHex(option:sub(2)) or ink
                button(pop, option, {
                    TextColor3 = tint, Size = UDim2.new(1, 0, 0, 30),
                    Position = UDim2.fromOffset(0, (index - 1) * 34),
                }, function() set(option) rebuild(true) end)
            end
        end)
        if key == "Color" then
            dropdown.Size = UDim2.fromScale(0.47, 1)
            local hex = box(area, "#RRGGBB", {Position = UDim2.fromScale(0.50, 0), Size = UDim2.fromScale(0.5, 1)})
            hex.Text = get()
            hex.FocusLost:Connect(function()
                if hex.Text:match("^#%x%x%x%x%x%x$") then set(hex.Text) rebuild(true)
                else hex.Text = get() end
            end)
        end
    elseif type(entry.default) == "boolean" then
        local toggle = button(area, "", full)
        local function refresh()
            toggle.Text = get() and "On" or "Off"
            toggle.TextColor3 = get() and mint or muted
        end
        toggle.Activated:Connect(function() set(not get()) refresh() end)
        refresh()
    elseif type(entry.default) == "number" then
        local track = round(new("Frame", area, {
            Size = UDim2.new(0.70, 0, 1, 0), BackgroundColor3 = control, Active = true,
        }))
        local fill = round(new("Frame", track, {BackgroundColor3 = mint, BackgroundTransparency = 0.54, BorderSizePixel = 0}))
        local number = box(area, "", {Position = UDim2.fromScale(0.73, 0), Size = UDim2.fromScale(0.27, 1)})
        local function refresh()
            local value = get()
            number.Text = string.format("%.3g", value)
            fill.Size = UDim2.fromScale((value - entry.min) / (entry.max - entry.min), 1)
        end
        track.InputBegan:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
            sliderInput = input
            slider = function(x)
                local alpha = math.clamp((x - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X), 0, 1)
                local value = entry.min + (entry.max - entry.min) * alpha
                local step = entry.max <= 5 and 100 or 1
                set(math.floor(value * step + 0.5) / step)
                refresh()
            end
            slider(input.Position.X)
        end)
        number.FocusLost:Connect(function()
            local value = tonumber(number.Text)
            if finite(value) then set(value) end
            refresh()
        end)
        refresh()
    else
        local settingBox = box(area, entry.default, full)
        settingBox.Text = get()
        settingBox.FocusLost:Connect(function()
            set(settingBox.Text)
            settingBox.Text = get()
        end)
    end
    if not mobile then
        card.MouseEnter:Connect(function() message(entry.help) end)
    end
end

            config.Mode, armed() and "armed" or "standby", rate, accuracy.damage, accuracy.received)

                item.owner and item.owner.DisplayName or item.model.Name, target.distance,
                target.counter, item.evasion, item.exposureMean)

            targetHud.Text = first

    end

        clearMarkers()

    end
    for _, item in ipairs(opponents) do
        local model, root = item.model, item.root
        if not markers[model] then

        marker.tag.Adornee = root

        if config.Health then table.insert(lines, string.format("%d HP", math.ceil(item.humanoid.Health))) end

            if finite(ammo) then table.insert(lines, string.format("%.0f ammo", ammo)) end
        end
        table.insert(lines, string.format("%d studs", math.floor(distance + 0.5)))
        marker.text.Text = table.concat(lines, " · ")
        marker.highlight.Enabled = config.Boxes and distance <= config.Range
        marker.highlight.FillColor, marker.highlight.OutlineColor = tint, tint
    end
    for model, marker in pairs(markers) do
        if not observations[model] then
            marker.tag:Destroy()
            marker.highlight:Destroy()
            markers[model] = nil
        end
    end
end

local function suggest(now, camera, solution)
    if config.Movement == "Off" or now - lastMove < 0.15 then return end
    local selfRoot = rootOf(player.Character)
    if not selfRoot then return end
    local action = config.Movement
    if action == "Adaptive" then
        if solution.distance < 12 then action = "Disengage"
        elseif solution.counter == "Sleepy" then action = "Peek"
        elseif solution.counter == "UE" then action = "Hold"
        else action = "Orbit" end
    end
    local position = selfRoot.Position
    local targetPosition = solution.item.root.Position
    if action == "Orbit" then
        local theta = now * config.OrbitSpeed
        position = Vector3.new(targetPosition.X, position.Y, targetPosition.Z)
            + Vector3.new(math.cos(theta), 0, math.sin(theta)) * config.OrbitRadius
    elseif action == "Disengage" then
        local away = position - targetPosition
        away = Vector3.new(away.X, 0, away.Z)
        if away.Magnitude > 0.01 then position = position + away.Unit * config.OrbitRadius end
    elseif action == "Peek" then
        local side = camera.CFrame.RightVector
        position = position + Vector3.new(side.X, 0, side.Z) * math.min(8, config.OrbitRadius)
    end
    lastMove = now
    strategy:Fire(action, solution.item.model, position)
end

local triggerParams = RaycastParams.new()
triggerParams.FilterType = Enum.RaycastFilterType.Exclude
triggerParams.IgnoreWater = true
local function requestFire(now, camera, weapon, solution)
    local char = player.Character
    local ammo = char:GetAttribute("AkiraAmmo")
    if char:GetAttribute("AkiraReloading") == true or (finite(ammo) and ammo <= 0) then return end
    local origin = camera.CFrame.Position
    local direction
    local kind
    if config.Mode == "Rage" and config.AutoFire and solution and now - targetSince >= solution.settle then
        direction = solution.point - origin
        kind = "Rage"
    elseif config.Mode == "Legit" and config.Trigger then
        triggerParams.FilterDescendantsInstances = {char, camera}
        local range = math.min(config.Range, weapon.WeaponRange)
        local ray = workspace:Raycast(origin, camera.CFrame.LookVector * range, triggerParams)
        local hit
        if ray then
            for _, item in ipairs(opponents) do
                if ray.Instance:IsDescendantOf(item.model) and item.model:GetAttribute("AkiraProtected") ~= true then
                    hit = item
                    break
                end
            end
        end
        if hit ~= triggerTarget then triggerTarget, triggerSince = hit, now end
        if hit and now - triggerSince >= config.TriggerDelay and now - lastRequest >= weapon.Interval then
            lastRequest = now
            fire:Fire("Trigger", camera.CFrame.LookVector, hit.model)
        end
        return
    end
    if direction and direction.Magnitude > 0.01 and now - lastRequest >= weapon.Interval then
        lastRequest = now
        fire:Fire(kind, direction.Unit, solution.item.model, solution.point)
    end
end

local fovCamera, originalFov, appliedFov
local function restoreFov()
    if fovCamera and fovCamera.FieldOfView == appliedFov then fovCamera.FieldOfView = originalFov end
    fovCamera, originalFov, appliedFov = nil, nil, nil
end

local renderName = "Akira.gg.Camera." .. JSON:GenerateGUID(false)
RunService:BindToRenderStep(renderName, Enum.RenderPriority.Camera.Value + 1, function(dt)
    if not alive then return end
    local camera = workspace.CurrentCamera
    local desired = allowed() and config.CameraFov >= 1 and math.clamp(config.CameraFov, 1, 110)
    if not desired or camera ~= fovCamera then restoreFov() end
    if camera and desired then
        if not fovCamera then fovCamera, originalFov = camera, camera.FieldOfView end
        appliedFov = desired
        if camera.FieldOfView ~= desired then camera.FieldOfView = desired end
    end
    if not camera or open or not armed() or not target then return end
    if (config.Mode == "Rage" and not config.Steer) or (config.Mode == "Legit" and config.Silent) then return end
    local from = camera.CFrame.Position
    if (target.point - from).Magnitude > 0.01 then
        local alpha = config.Mode == "Rage" and 1 or 1 - math.exp(-config.Smooth * dt)
        camera.CFrame = camera.CFrame:Lerp(CFrame.lookAt(from, target.point), alpha)
    end
end)

local tickAt, visualAt = 0, 0
local lastWeaponKey, lastMode
connect(RunService.Heartbeat, function()
    if not alive then return end
    local now = os.clock()
    if now >= due and revision ~= savedRevision and config.AutoSave then save(false) end
    if now < tickAt then return end
    tickAt = now + 1 / 24
    local camera = workspace.CurrentCamera
    if allowed() then
        updateOpponents(now)
    else
        table.clear(opponents)
        table.clear(observations)
    end
    local weaponKey = currentWeapon()
    if weaponKey ~= lastWeaponKey or config.Mode ~= lastMode then
        lastWeaponKey, lastMode = weaponKey, config.Mode
        target, triggerTarget = nil, nil
    end
    local weapon = weaponSettings()
    if camera and not open and armed() and weapon.Enabled then
        local solution = acquire(camera, camera.CFrame.Position, weapon, now, true)
        if not target or not solution or solution.item ~= target.item or solution.visible ~= target.visible then
            targetSince = now
        end
        target = solution
        requestFire(now, camera, weapon, solution)
        if config.Mode == "Rage" and solution then suggest(now, camera, solution) end
    else
        target, triggerTarget, targetSince = nil, nil, now
    end
    if now >= visualAt then
        visualAt = now + 0.1
        updateVisuals(camera, color())
    end
end)

local function shutdown()
    if not alive then return end
    alive = false
    RunService:UnbindFromRenderStep(renderName)
    restoreFov()
    if config.AutoSave then save(false) end
    for _, connection in ipairs(connections) do connection:Disconnect() end
    clearMarkers()
end
connect(stop.Event, shutdown)
connect(gui.Destroying, shutdown)
connect(script.Destroying, function()
    shutdown()
    gui:Destroy()
end)

if type(getgenv) == "function" then
    local ok, environment = pcall(getgenv)
    if ok and type(environment) == "table" then
        local handle = {
            Open = function()
                if alive and allowed() then setOpen(true) end
            end,
            Close = function()
                if alive then setOpen(false) end
            end,
            Unload = function()
                if not alive then return end
                shutdown()
                gui:Destroy()
            end,
        }
        environment.AkiraGG = handle
        gui.Destroying:Once(function()
            if environment.AkiraGG == handle then environment.AkiraGG = nil end
        end)
    end
end
